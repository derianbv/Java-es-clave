List<String> nombres = Arrays.asList("Ana", "Juan", "Pedro", "María", "Luis");

long count = nombres.stream()  // Crear Stream desde la lista
    .filter(n -> n.length() > 4)  // Operación intermedia: filtrar nombres con más de 4 letras
    .count();  // Operación terminal: contar elementos que cumplen el filtro

System.out.println("Nombres con más de 4 letras: " + count);
